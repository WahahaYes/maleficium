\begin{abstract}
One source that compiles to a PDF today and is meant to become an
interactive HTML paper later. Every widget macro the package offers
appears at least once, beside plain figures, subfigures, math,
citations, cross-references and a theorem.
\end{abstract}

\section{Introduction}
Text typeset with \TeX{}~\cite{knuth1984texbook} and \LaTeX{}~\cite{lamport1994latex}
reads the same in print and on screen; the widgets below add an
interactive layer on top of that~\cite{shannon1948mathematical}.
Section~\ref{sec:guide} explains each widget, and
Figures~\ref{fig:model}, \ref{fig:video}, \ref{fig:chart} and~\ref{fig:html},
Table~\ref{tab:results} and Figures~\ref{fig:subs} and~\ref{fig:pair} show them.

\section{Math and theorems}
Entropy is defined in Equation~\eqref{eq:entropy} and bounded in
Theorem~\ref{thm:bound}.
\begin{equation}\label{eq:entropy}
H(X) = -\sum_{i=1}^{n} p_i \log p_i .
\end{equation}
\begin{theorem}\label{thm:bound}
For a distribution on $n$ outcomes, $0 \le H(X) \le \log n$.
\end{theorem}
\begin{proof}
Jensen's inequality gives $H(X) \le \log \sum_i p_i \cdot \tfrac{1}{p_i} = \log n$,
and each term of Equation~\eqref{eq:entropy} is nonnegative.
\end{proof}

\section{What each widget does}\label{sec:guide}
\begin{description}
\item[Model] \verb|\interactivemodel| loads a glTF model (\texttt{.glb}) for orbiting
  and zooming. The PDF shows the poster (Figure~\ref{fig:model}).
\item[Video] \verb|\interactivevideo| plays a clip, with an optional remote URL
  and checksum for large files. The PDF shows the poster frame (Figure~\ref{fig:video}).
\item[Table] \verb|\interactivetable| reads a CSV into a sortable table. The PDF
  prints only the first \texttt{pdfrows} rows (Table~\ref{tab:results}).
\item[Chart] \verb|\interactivechart| renders a Vega-Lite specification with
  hover details; the PDF shows a poster (Figure~\ref{fig:chart}).
\item[HTML] \verb|\interactive| embeds a folder with an \texttt{index.html} in a
  sandboxed frame (Figure~\ref{fig:html}).
\end{description}
Every widget requires \texttt{alt=} text, which becomes the image
description in the HTML view. The app renders the PDF's picture of a
model, chart or HTML widget from the widget itself, so these carry no
\texttt{poster=}; a video needs a \texttt{poster=} frame until video
posters are rendered.
